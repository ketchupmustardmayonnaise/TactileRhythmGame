./config/systemd/user/audio-service.service
---------
[Unit]
Description=My Custom Audio User Service
After=sound.target

[Service]
Type=simple
ExecStartPre=/usr/bin/sleep 3
ExecStartPre=/usr/bin/amixer -c 0 sset 'Headphone' 115
ExecStart=/home/muon/bin/audio-service
Environment=TTS_ASSET_DIR=/home/muon/tts-engine/assets
Restart=on-failure

[Install]
WantedBy=default.target
